\section{Datos y antecedentes}

% 2. Arbués explica la fuente y el punto de partida.
\begin{frame}{Fuente y motivo del análisis}
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
      \begin{itemize}
        \item MINCETUR publica los indicadores en la Plataforma Nacional de Datos Abiertos.
        \item La encuesta registra capacidad y uso de hospedajes a nivel nacional y regional.
        \item El MEF usa estos indicadores como referencia para estimar demanda turística.
      \end{itemize}
    \column{0.38\textwidth}
      \begin{block}{Lo encontrado}
        Los estudios previos se concentran en una región y usan métodos estadísticos clásicos.
      \end{block}
  \end{columns}
  \vfill
  {\tiny Fuentes: MINCETUR (2021), MEF (2021), Taipe y Laura (2021), Ñahuincopa y Rojas (2022).}
  \note{Arbués: 55 s. Explicar que el dataset sí se ha usado, pero no se encontró un trabajo con Hadoop o aprendizaje automático aplicado directamente a este CSV.}
\end{frame}

\begin{frame}{Columnas del dataset}
\scriptsize
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\textbf{Identificación y dimensiones}
\begin{itemize}
  \item \texttt{FECHA_CORTE}: fecha de actualización.
  \item \texttt{ANIO}, \texttt{MES}: periodo observado.
  \item \texttt{ID_CLASE}, \texttt{CLASE}: tipo de establecimiento.
  \item \texttt{ID_CATEGORIA}, \texttt{CATEGORIA}: categoría.
  \item \texttt{DEPARTAMENTO}, \texttt{ID_UBIGEO}: ubicación.
\end{itemize}
\textbf{Capacidad}
\begin{itemize}
  \item \texttt{NUMERO_ESTABLECIMIENTOS}: locales reportados.
  \item \texttt{NUMERO_HABITACIONES}: habitaciones disponibles.
  \item \texttt{NUMERO_PLAZAS_CAMA}: camas ofertadas.
\end{itemize}
\column{0.49\textwidth}
\textbf{Indicadores}
\begin{itemize}
  \item \texttt{PORCENTAJE_TNOH}: ocupación neta de habitaciones.
  \item \texttt{PORCENTAJE_TNOC}: ocupación neta de camas.
  \item \texttt{PROMEDIO_PERMANENCIA}: días promedio de estadía.
  \item \texttt{PROMEDIO_PERMANENCIA_NAC/EXT}: estadía nacional o extranjera.
\end{itemize}
\textbf{Movimientos y empleo}
\begin{itemize}
  \item \texttt{TOTAL_ARRIBOS}, \texttt{..._NAC}, \texttt{..._EXT}: llegadas.
  \item \texttt{TOTAL_PERNOCT}, \texttt{..._NAC}, \texttt{..._EXT}: noches.
  \item \texttt{TOTAL_EMPLEO}: personas empleadas reportadas.
\end{itemize}
\vfill
\textbf{Siglas:} TNOH es ocupación neta de habitaciones; TNOC, de camas. TT/TT significa total de clase y total de categoría, usado para evitar doble conteo.
\end{columns}
\end{frame}

\section{Consultas MapReduce}

\begin{frame}{Q01 — arribos por origen}
\small
\textbf{Ítem 1: consulta con dos o más campos.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿cómo evolucionaron los arribos nacionales y extranjeros por año?
  \item \textbf{Campos:} año, arribos nacionales y arribos extranjeros.
  \item \textbf{Resultado:} nacionales: 54,9 millones en 2019 y 34,6 millones en 2020.
\end{itemize}
\end{frame}

\begin{frame}{Q02 — ocupación por departamento}
\small
\textbf{Ítem 1: consulta con dos o más campos.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿cuál fue la TNOH promedio de cada departamento por año?
  \item TNOH significa ocupación neta de habitaciones.
  \item \textbf{Resultado:} Callao llegó a 41,5950 en 2024; Loreto, a 12,9917.
\end{itemize}
\end{frame}

\begin{frame}{Q03 — ocupación por clase}
\small
\textbf{Ítem 1: consulta con dos o más campos.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿cómo cambian TNOH y TNOC según la clase?
  \item TNOC significa ocupación neta de camas.
  \item \textbf{Resultado:} en 2024, Resort obtuvo TNOH 46,7742 y TNOC 48,2812.
\end{itemize}
\end{frame}

\begin{frame}{Q04 — empleo por capacidad}
\small
\textbf{Ítem 1: consulta con dos o más campos.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿cuántos empleos se reportan por cada 100 habitaciones?
  \item \textbf{Campos:} año, departamento, empleo y habitaciones.
  \item \textbf{Resultado:} Lima obtuvo 54,3562 empleos por cada 100 habitaciones en 2024.
\end{itemize}
\end{frame}

\begin{frame}{Q05 — estacionalidad mensual}
\small
\textbf{Ítem 1: consulta con dos o más campos.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿qué meses concentran más arribos y pernoctaciones?
  \item \textbf{Campos:} mes, arribos y pernoctaciones.
  \item \textbf{Resultado:} enero y febrero tuvieron los mayores acumulados.
\end{itemize}
\end{frame}


\section{Estadística y comparación}

\begin{frame}{Q06 — estadística de TNOH}
\small
\textbf{Ítem 2: promedio, mediana y desviación.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿cuál es el nivel central y la dispersión de la ocupación?
  \item \textbf{Resultado:} 1\,950 valores; media 21,7433; mediana 21,3400; desviación 7,0679.
  \item La mediana divide la mitad inferior y superior de los valores.
\end{itemize}
\end{frame}

\begin{frame}{Q07 — búsqueda de texto}
\small
\textbf{Ítem 3: búsqueda de subtexto.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿qué registros contienen la palabra hotel?
  \item \textbf{Campos:} clase, categoría y departamento.
  \item \textbf{Resultado:} 11\,525 coincidencias con el registro completo.
\end{itemize}
\end{frame}

\begin{frame}{Q08 — valores extremos}
\small
\textbf{Ítem 4: mayor y menor valor agrupado.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿qué departamento tuvo la menor y mayor TNOH media por año?
  \item \textbf{Resultado 2024:} Loreto fue el mínimo (12,9917) y Callao el máximo (41,5950).
\end{itemize}
\end{frame}

\begin{frame}{Q09 — cambio entre 2019 y 2024}
\small
\textbf{Ítem 5: dos MapReduce enlazados.}
\begin{itemize}
  \item Job 1 calcula la TNOH promedio por año y departamento.
  \item Job 2 une 2019 con 2024 y calcula la diferencia.
  \item \textbf{Resultado:} Apurímac tuvo la mayor caída: -18,7325 puntos porcentuales.
\end{itemize}
\end{frame}

\begin{frame}{Q10 — participación extranjera en Cusco}
\small
\textbf{Ítem 5: dos MapReduce enlazados.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿cómo cambió la proporción de arribos extranjeros?
  \item \textbf{Resultado:} 63,3747\% antes de la pandemia; 20,5234\% durante el impacto; 58,3507\% en recuperación.
\end{itemize}
\end{frame}

\section{Modelos}

\begin{frame}{Q11 — Gaussian Naive Bayes}
\small
\textbf{Ítem 6: modelo de clasificación.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿la ocupación mensual es alta o baja?
  \item Alta significa TNOH mayor o igual a la mediana del entrenamiento.
  \item \textbf{F1:} 0,4403 en 2024 y 0,4675 en 2025-I.
\end{itemize}
\end{frame}

\begin{frame}{Q12 — regresión logística}
\small
\textbf{Ítem 6: modelo de clasificación.}
\begin{itemize}
  \item Se respondió la misma pregunta con las mismas variables.
  \item \textbf{F1:} 0,7931 en 2024 y 0,6939 en 2025-I.
  \item Fue el clasificador más equilibrado.
\end{itemize}
\end{frame}

\begin{frame}{Q13 — regresión lineal}
\small
\textbf{Ítem 7: modelo de regresión.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿qué TNOH se predice usando los demás indicadores?
  \item Entrenamiento: 2019--2023; evaluación: 2024 y 2025-I.
  \item \textbf{Resultado 2024:} MAE 3,1922; RMSE 4,0920; R cuadrado 0,6435.
\end{itemize}
\end{frame}

\begin{frame}{Q14 — regresión ridge}
\small
\textbf{Ítem 7: modelo de regresión.}
\begin{itemize}
  \item \textbf{Pregunta:} ¿la regularización mejora la predicción de TNOH?
  \item \textbf{Resultado 2025-I:} MAE 3,0133; RMSE 3,8796; R cuadrado 0,6703.
  \item La mejora frente al modelo lineal fue pequeña.
\end{itemize}
\end{frame}

% 10. Ambos cierran con lo demostrado y los límites.
\begin{frame}{Conclusiones y límites}
  \begin{itemize}
    \item Se logró unificar siete CSV sin modificar los originales.
    \item Las catorce consultas dejaron salida real y un JAR reproducible.
    \item La investigación previa ayudó a normalizar el esquema, evitar doble conteo y elegir una evaluación temporal.
    \item La regresión logística obtuvo el mejor F1 y ridge mejoró ligeramente el RMSE.
  \end{itemize}
  \begin{alertblock}{Límites}
    2025 contiene solo seis meses. El clúster tuvo un nodo y el dataset no representa macrodatos por volumen.
  \end{alertblock}
